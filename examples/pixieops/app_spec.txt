<project_specification>
  <project_name>PixieOps - AI Workforce Planning Assistant</project_name>

  <overview>
    Build an AI-powered workforce planning assistant that helps find employees for projects via Slack.
    Built on AWS Bedrock AgentCore with S3 Vectors for cost-effective vector storage, it delivers
    intelligent employee search across skills, certifications, roles, and availability data.
    The system uses bidirectional search across Knowledge Base, Snowflake, and utilization data
    to provide comprehensive employee profiles with real-time availability forecasts.

    Vision: An intelligent Slack assistant that instantly finds the right employees for any IT project
    by semantically searching skills, certifications, and experience while providing real-time
    availability forecasts - powered by AWS Bedrock AgentCore with enterprise-grade observability
    and continuous quality evaluation.
  </overview>

  <intent_classification_schema>
    <description>
      LLM-powered query analysis using Claude Haiku (~500ms latency, ~$0.0003 per query).
      Runs ONCE before agent execution to guide tool selection.
    </description>

    <intent_types>
      <role>Job title searches (e.g., "Principal Architect", "Senior Engineer")</role>
      <skill>Technical skill searches (e.g., "AWS certified", "Python experience")</skill>
      <industry>Industry experience (e.g., "healthcare", "financial services")</industry>
      <certification>Professional certifications (e.g., "AWS Certified", "PMP")</certification>
      <specific_person_availability>Asking about a SPECIFIC NAMED person's availability (e.g., "availability of John Smith")</specific_person_availability>
      <availability>GENERAL availability requirements without specific person (e.g., "who is free in Q1")</availability>
    </intent_types>

    <output_schema>
      {
        "intent_types": ["role", "skill", "industry", "certification", "specific_person_availability", "availability"],
        "entities": {
          "roles": ["Principal Solution Architect", "Senior Engineer"],
          "skills": ["AWS", "Python", "Kubernetes"],
          "industries": ["healthcare", "financial services"],
          "certifications": ["AWS Certified Solutions Architect", "PMP"],
          "person_name": "John Smith",
          "time_period": "Q1 2025"
        },
        "search_guidance": "role_only | skill_only | comprehensive | availability_only",
        "availability_filter": true,
        "confidence": 0.95
      }
    </output_schema>

    <search_guidance_rules>
      <role_only>Pure role/title search with NO other criteria → use Snowflake as primary</role_only>
      <skill_only>ONLY skills, certifications, or industry experience → use Knowledge Base as primary</skill_only>
      <comprehensive>Multiple intent types (combination query) → search both sources</comprehensive>
      <availability_only>Specific person's availability → go directly to utilization tool</availability_only>
    </search_guidance_rules>

    <default_fallback>
      When intent classification fails, use default:
      - intent_types: []
      - search_guidance: "comprehensive"
      - confidence: 0.3 (low confidence indicates fallback)
      - availability_filter: false
    </default_fallback>

    <time_period_parsing>
      - Assume current/next for ambiguous periods
      - "Q1" = next Q1 from current date
      - "next month" = from today
      - "December" = next December if current month is past December
    </time_period_parsing>
  </intent_classification_schema>

  <knowledge_base_retrieval>
    <description>
      Semantic search with LLM verification to ensure accuracy for combination queries.
    </description>

    <retrieval_process>
      1. Perform raw retrieval from Bedrock KB (HYBRID search, up to 100 results)
      2. Filter by relevance score threshold (0.30 minimum)
      3. For each candidate, run LLM verification to confirm match
      4. Return only verified matches with bio summary, role, relevance score
    </retrieval_process>

    <relevance_threshold>0.30 - lower threshold to get more candidates, LLM verification ensures accuracy</relevance_threshold>

    <llm_verification>
      <purpose>Verify that retrieved bio ACTUALLY matches ALL query criteria</purpose>
      <model>Claude 3.5 Sonnet (anthropic.claude-3-5-sonnet-20241022-v2:0)</model>
      <verification_rules>
        - For combination queries, person MUST match ALL criteria
        - Check each criterion type: role, certification, skill, industry
        - Answer YES only if bio clearly demonstrates ALL requested qualifications
        - Answer NO if ANY criterion is missing or unclear
      </verification_rules>
      <fail_open_behavior>On verification error, include result by default (fail open)</fail_open_behavior>
      <unclear_response_handling>If LLM response is neither YES nor NO, include by default with warning</unclear_response_handling>
    </llm_verification>

    <certification_matching>
      - Look for exact certification names
      - Match abbreviations (CSM, PSM, PMP, CKA, CKAD, TOGAF, ITIL)
      - Check certifications section in bio
      - Broad recognition: any certification mentioned in bios is searchable
    </certification_matching>

    <employee_name_extraction>
      - Extract from S3 filename pattern: "Ahmed Suliman - Bio.pdf"
      - Regex: ^(.+?)\s*[-–]\s* to extract name before dash
      - Fallback: remove ".pdf" and " Bio" from filename
    </employee_name_extraction>

    <employee_name_matching>
      - Use fuzzy matching for misspelled names
      - Suggest closest matches: "Did you mean John Smith?"
      - Case-insensitive comparison
    </employee_name_matching>

    <result_limits>
      - Return top 10 most relevant results
      - If more found, state "X more employees found, please narrow your search"
    </result_limits>

    <mixed_criteria_handling>
      - Search valid criteria only
      - Silently ignore unrecognized criteria (e.g., "unicorn experience")
      - Do not warn user about ignored criteria
    </mixed_criteria_handling>
  </knowledge_base_retrieval>

  <data_source_precedence>
    <description>
      Rules for resolving conflicts when employee data exists in multiple sources.
    </description>

    <source_of_truth>
      Snowflake is the PRIMARY source of truth for:
      - Current job title/role (BUSINESS_TITLE)
      - Practice area (PRACTICE)
      - Email address (WORK_EMAIL)
      - Employment status (active vs terminated)
    </source_of_truth>

    <secondary_source>
      Knowledge Base (bios) is SECONDARY source for:
      - Skills and technologies
      - Certifications
      - Project history
      - Industry experience
      - Detailed background
    </secondary_source>

    <cross_reference_scenarios>
      <both_sources>
        When employee found in BOTH sources:
        - data_source: "both"
        - Use Snowflake for role, practice, email
        - Use KB for bio_summary, skills, certifications
        - This is the IDEAL scenario
      </both_sources>

      <snowflake_only>
        When employee found in Snowflake but NOT in KB:
        - data_source: "snowflake"
        - bio_summary: "No bio available"
        - State in response: "Bio not available for [name]"
      </snowflake_only>

      <kb_only>
        When employee found in KB but NOT in Snowflake:
        - data_source: "knowledge_base"
        - role: Extract from bio if possible, else "Unknown"
        - practice: "Unknown"
        - email: "Unknown"
        - State in response: "Note: [name]'s role data may be outdated"
      </kb_only>

      <neither_source>
        When employee not found in either source:
        - Return: "No employees found matching [criteria]"
      </neither_source>
    </cross_reference_scenarios>

    <merge_priority>
      When merging results from both sources:
      1. Start with KB results (bio data)
      2. Overlay Snowflake data (takes precedence for role/practice/email)
      3. Mark data_source appropriately
    </merge_priority>
  </data_source_precedence>

    <description>
      Graceful degradation strategies for various failure scenarios.

      - Continue processing (do not block user)

  <example_prompts>
    <intent_classifier_prompt>
      You are a query classifier for an IT workforce management system.
      Analyze the user's query and extract structured information to guide employee search.

      INTENT TYPES:
      - role: Job title searches (e.g., "Principal Architect", "Senior Engineer")
      - skill: Technical skill searches (e.g., "AWS certified", "Python experience")
      - industry: Industry experience (e.g., "healthcare", "financial services")
      - certification: Professional certifications (e.g., "AWS Certified", "PMP")
      - specific_person_availability: Asking about a specific named person's availability
      - availability: General availability requirements (time periods, utilization)

      IMPORTANT - COMBINATION QUERIES:
      - A query can have MULTIPLE intent types
      - Extract ALL intent types present in the query
      - List ALL entities found for each category
      - Examples:
        * "AWS certified Solution Architects" → intent_types: ["certification", "role"]
